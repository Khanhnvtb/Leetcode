//https://leetcode.com/problems/sort-colors/description/

class Solution {
public:
    void sortColors(vector<int>& nums) {
        int used[3] = {0};
        for(int i = 0; i < nums.size(); i++) used[nums[i]]++;
        int i = 0;
        for(int j = 0; j < 3; j++){
            while(used[j]){
                nums[i++] = j;
                used[j]--;
            }
        }
    }
};
