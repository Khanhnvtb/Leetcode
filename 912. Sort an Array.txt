//https://leetcode.com/problems/sort-an-array/description/

class Solution {
public:
    vector<int> sortArray(vector<int>& nums) {
        int used[50005] = {0}, neg_used[50005] = {0};
        for(int i = 0; i < nums.size(); i++){
            if(nums[i] >= 0) used[nums[i]]++;
            else neg_used[-nums[i]]++;
        } 
        vector<int> ans;
        for(int i = -50004; i < 0; i++){
            while(neg_used[-i]){
                ans.push_back(i);
                neg_used[-i]--;
            }
        }
        for(int i = 0; i < 50005; i++){
            while(used[i]){
                ans.push_back(i);
                used[i]--;
            }
        }
        return ans;
    }
};
