//https://leetcode.com/problems/merge-intervals/description/

class Solution {
public:
    vector<vector<int>> merge(vector<vector<int>>& intervals) {
        vector<vector<int>> ans;
        sort(intervals.begin(), intervals.end());
        int i = 0, size = intervals.size() - 1, j, max_value = intervals[i][1];
        vector<int> v = {intervals[i][0]};
        while(i < size){
            j = i + 1;
            if(intervals[j][0] > max_value){
                v.push_back(max_value);
                ans.push_back(v);
                v = {intervals[j][0]};
                max_value = intervals[j][1];
            }else if(intervals[j][1] > max_value) max_value = intervals[j][1];
            i++;
        }
        v.push_back(max_value);
        ans.push_back(v);
        return ans;
    }
};
